Lazy options

At a glance

Type
New feature
Lazy options added
2.1

Since symfony/options-resolver 2.1, a default value can be a closure typed with Options, evaluated when the option is read.

Lazy default value

Other option read

A closure whose first argument is type-hinted Options is only called when the option is not given, and can read the other options.

PHP 5.3
use Symfony\Component\OptionsResolver\Options;
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = new OptionsResolver();
$resolver
    ->setRequired(array('firstName'))
    ->setDefaults(array(
        'key' => function (Options $options) {
            return 'foo' === $options['firstName'] ? 'bar' : 'baz';
        },
    ));

// array('key' => 'bar', 'firstName' => 'foo')
$options = $resolver->resolve(array('firstName' => 'foo'));
use Symfony\Component\OptionsResolver\Options;
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = (new OptionsResolver())
    ->setRequired(['firstName'])
    ->setDefaults([
        'key' => function (Options $options) {
            return 'foo' === $options['firstName'] ? 'bar' : 'baz';
        },
    ]);

// ['key' => 'bar', 'firstName' => 'foo']
$options = $resolver->resolve(['firstName' => 'foo']);
use Symfony\Component\OptionsResolver\Options;
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = (new OptionsResolver())
    ->setRequired(['firstName'])
    ->setDefaults([
        'key' => fn (Options $options): string => 'foo' === $options['firstName'] ? 'bar' : 'baz',
    ]);

// ['key' => 'bar', 'firstName' => 'foo']
$options = $resolver->resolve(['firstName' => 'foo']);
use Symfony\Component\OptionsResolver\Options;
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = new OptionsResolver()
    ->setRequired(['firstName'])
    ->setDefaults([
        'key' => fn (Options $options): string => 'foo' === $options['firstName'] ? 'bar' : 'baz',
    ]);

// ['key' => 'bar', 'firstName' => 'foo']
$options = $resolver->resolve(['firstName' => 'foo']);

Previous default value

A second argument of the closure, $previousValue, receives the default value set before.

  • OptionsResolverInterface::setDefaults() in v2.1.0
  • OptionsResolver::setDefaults() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\Options;
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = new OptionsResolver();
$resolver
    ->setDefaults(array('roles' => array('foo')))
    ->setDefaults(array(
        'roles' => function (Options $options, array $previousValue) {
            return array_merge($previousValue, array('bar'));
        },
    ));

// array('roles' => array('foo', 'bar'))
$options = $resolver->resolve();
use Symfony\Component\OptionsResolver\Options;
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = (new OptionsResolver())
    ->setDefaults(['roles' => ['foo']])
    ->setDefaults([
        'roles' => function (Options $options, array $previousValue) {
            return array_merge($previousValue, ['bar']);
        },
    ]);

// ['roles' => ['foo', 'bar']]
$options = $resolver->resolve();
use Symfony\Component\OptionsResolver\Options;
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = (new OptionsResolver())
    ->setDefaults(['roles' => ['foo']])
    ->setDefaults([
        'roles' => fn (Options $options, array $previousValue): array => array_merge($previousValue, ['bar']),
    ]);

// ['roles' => ['foo', 'bar']]
$options = $resolver->resolve();
