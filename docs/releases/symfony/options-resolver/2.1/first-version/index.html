First version: OptionsResolver and Options

At a glance.

Type
New feature
Package created
2.1

symfony/options-resolver 2.1 is the first version of the package: OptionsResolver checks an array of options against default, optional and required options and allowed values, and Options gives lazy options, whose default value depends on other options.

Minimal configuration

No option

PHP 5.3
$resolver = new OptionsResolver();

// array()
$options = $resolver->resolve();

Unknown option

PHP 5.3
$resolver = new OptionsResolver();

// InvalidOptionsException: The option "firstName" does not exist. Known options are: ""
$options = $resolver->resolve(array('firstName' => 'Jane'));

Required option setRequired()

Option given

PHP 5.3
$resolver = new OptionsResolver();
$resolver->setRequired(array('firstName'));

// array('firstName' => 'Jane')
$options = $resolver->resolve(array('firstName' => 'Jane'));

Option missing

PHP 5.3
$resolver = new OptionsResolver();
$resolver->setRequired(array('firstName'));

// MissingOptionsException: The required option "firstName" is  missing.
$options = $resolver->resolve();

Default value setDefaults()

Option not given

PHP 5.3
$resolver = new OptionsResolver();
$resolver->setDefaults(array('gender' => 'female'));

// array('gender' => 'female')
$options = $resolver->resolve();

Option given

PHP 5.3
$resolver = new OptionsResolver();
$resolver->setDefaults(array('gender' => 'female'));

// array('gender' => 'male')
$options = $resolver->resolve(array('gender' => 'male'));

Lazy default value

A closure whose first argument is type-hinted Options is only called when the option is not given, and can read the other options.

PHP 5.3
$resolver = new OptionsResolver();
$resolver
    ->setRequired(array('firstName'))
    ->setDefaults(array(
        'gender' => function (Options $options) {
            return 'John' === $options['firstName'] ? 'male' : 'female';
        },
    ));

// array('gender' => 'male', 'firstName' => 'John')
$options = $resolver->resolve(array('firstName' => 'John'));

Previous default value

A second argument of the closure, $previousValue, receives the default value set before.

PHP 5.3
$resolver = new OptionsResolver();
$resolver
    ->setDefaults(array('roles' => array('ROLE_USER')))
    ->setDefaults(array(
        'roles' => function (Options $options, $previousValue) {
            return array_merge($previousValue, array('ROLE_ADMIN'));
        },
    ));

// array('roles' => array('ROLE_USER', 'ROLE_ADMIN'))
$options = $resolver->resolve();

Replaced default value

replaceDefaults() erases the previous default value: $previousValue is null.

PHP 5.3
$resolver = new OptionsResolver();
$resolver
    ->setDefaults(array('roles' => array('ROLE_USER')))
    ->replaceDefaults(array(
        'roles' => function (Options $options, $previousValue) {
            return $previousValue;
        },
    ));

// array('roles' => null)
$options = $resolver->resolve();

Optional option setOptional()

Option not given

PHP 5.3
$resolver = new OptionsResolver();
$resolver->setOptional(array('email'));

// array()
$options = $resolver->resolve();

Option given

PHP 5.3
$resolver = new OptionsResolver();
$resolver->setOptional(array('email'));

// array('email' => 'jane@example.com')
$options = $resolver->resolve(array('email' => 'jane@example.com'));

Allowed values setAllowedValues()

Allowed value

PHP 5.3
$resolver = new OptionsResolver();
$resolver
    ->setDefaults(array('gender' => 'female'))
    ->setAllowedValues(array('gender' => array('male', 'female')));

// array('gender' => 'male')
$options = $resolver->resolve(array('gender' => 'male'));

Value not allowed

PHP 5.3
$resolver = new OptionsResolver();
$resolver
    ->setDefaults(array('gender' => 'female'))
    ->setAllowedValues(array('gender' => array('male', 'female')));

// InvalidOptionsException: The option "gender" has the value "other", but is expected to be one of "male", "female"
$options = $resolver->resolve(array('gender' => 'other'));

Added allowed value

PHP 5.3
$resolver = new OptionsResolver();
$resolver
    ->setDefaults(array('gender' => 'female'))
    ->setAllowedValues(array('gender' => array('male', 'female')))
    ->addAllowedValues(array('gender' => array('other')));

// array('gender' => 'other')
$options = $resolver->resolve(array('gender' => 'other'));
