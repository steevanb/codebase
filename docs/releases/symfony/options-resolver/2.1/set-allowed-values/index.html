setAllowedValues()

At a glance

Type
New feature
setAllowedValues() added
2.1

Since symfony/options-resolver 2.1, setAllowedValues() restricts the values of an option to a list.

Allowed values setAllowedValues()

Allowed value

  • OptionsResolverInterface::setAllowedValues() in v2.1.0
  • OptionsResolver::setAllowedValues() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = new OptionsResolver();
$resolver
    ->setDefaults(array('key' => 'female'))
    ->setAllowedValues(array('key' => array('male', 'female')));

// array('key' => 'male')
$options = $resolver->resolve(array('key' => 'male'));
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = (new OptionsResolver())
    ->setDefaults(['key' => 'female'])
    ->setAllowedValues(['key' => ['male', 'female']]);

// ['key' => 'male']
$options = $resolver->resolve(['key' => 'male']);
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = new OptionsResolver()
    ->setDefaults(['key' => 'female'])
    ->setAllowedValues(['key' => ['male', 'female']]);

// ['key' => 'male']
$options = $resolver->resolve(['key' => 'male']);
