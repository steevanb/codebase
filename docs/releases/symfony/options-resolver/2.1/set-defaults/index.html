setDefaults(array $defaultValues)

At a glance

Type
New feature
setDefaults() added
2.1

Since symfony/options-resolver 2.1, setDefaults() defines options with a default value, used when the option is not given, lazy when it is a closure typed with Options.

Default value

Null value

  • Symfony 2.x documentation: Usage
  • OptionsResolverInterface::setDefaults() in v2.1.0
  • OptionsResolver::setDefaults() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = new OptionsResolver();
$resolver->setDefaults(array('key' => null));

// array('key' => null)
$options = $resolver->resolve();
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = (new OptionsResolver())
    ->setDefaults(['key' => null]);

// ['key' => null]
$options = $resolver->resolve();
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = new OptionsResolver()
    ->setDefaults(['key' => null]);

// ['key' => null]
$options = $resolver->resolve();

Lazy default value

A closure whose first argument is type-hinted Options, $options, is only called when the option is read, and can read the other options.

Its optional second argument, $previousValue, receives the default value set before.

Other option read

  • Symfony 2.x documentation: Default Values that Depend on another Option
  • OptionsResolverInterface::setDefaults() in v2.1.0
  • OptionsResolver::setDefaults() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\Options;
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = new OptionsResolver();
$resolver
    ->setRequired(array('firstName'))
    ->setDefaults(array(
        'key' => function (Options $options) {
            return strtoupper($options['firstName']);
        },
    ));

// array('key' => 'FOO', 'firstName' => 'foo')
$options = $resolver->resolve(array('firstName' => 'foo'));
use Symfony\Component\OptionsResolver\Options;
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = (new OptionsResolver())
    ->setRequired(['firstName'])
    ->setDefaults([
        'key' => function (Options $options) {
            return strtoupper($options['firstName']);
        },
    ]);

// ['key' => 'FOO', 'firstName' => 'foo']
$options = $resolver->resolve(['firstName' => 'foo']);
use Symfony\Component\OptionsResolver\Options;
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = (new OptionsResolver())
    ->setRequired(['firstName'])
    ->setDefaults([
        'key' => fn (Options $options): string => strtoupper($options['firstName']),
    ]);

// ['key' => 'FOO', 'firstName' => 'foo']
$options = $resolver->resolve(['firstName' => 'foo']);

use Symfony\Component\OptionsResolver\Options;

$resolver = new OptionsResolver();
$resolver
    ->setDefaults(array('roles' => array('foo')))
    ->setDefaults(array(
        'roles' => function (Options $options, array $previousValue) {

$options = $resolver->resolve();

Closure not typed with Options

  • OptionsResolverInterface::setDefaults() in v2.1.0
  • OptionsResolver::setDefaults() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = new OptionsResolver();
$resolver->setDefaults(array(
    // A closure whose first argument, $options, is not type-hinted Options is not lazy:
    // it is the value of the option
    'key' => function ($options) {
        return 'foo';
    },
));

// array('key' => Closure)
$options = $resolver->resolve();
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = (new OptionsResolver())
    ->setDefaults([
        // A closure whose first argument, $options, is not type-hinted Options is not lazy:
        // it is the value of the option
        'key' => function ($options) {
            return 'foo';
        },
    ]);

// ['key' => Closure]
$options = $resolver->resolve();
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = (new OptionsResolver())
    ->setDefaults([
        // A closure whose first argument, $options, is not type-hinted Options is not lazy:
        // it is the value of the option
        'key' => fn ($options): string => 'foo',
    ]);

// ['key' => Closure]
$options = $resolver->resolve();
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = new OptionsResolver()
    ->setDefaults([
        // A closure whose first argument, $options, is not type-hinted Options is not lazy:
        // it is the value of the option
        'key' => fn (mixed $options): string => 'foo',
    ]);

// ['key' => Closure]
$options = $resolver->resolve();
