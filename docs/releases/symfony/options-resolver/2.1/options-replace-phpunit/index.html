Options::replace(array $options)

At a glance

Type
New feature
replace() added
2.1

Since symfony/options-resolver 2.1, Options::replace() replaces all the options by the given ones: the results of the examples are shown with PHPUnit assertions.

Replace all options

Options replaced

  • Declaration of Options::replace() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->set('firstName', 'foo');
$options->replace(array('key' => 'bar'));

static::assertSame(array('key' => 'bar'), $options->all());
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->set('firstName', 'foo');
$options->replace(['key' => 'bar']);

static::assertSame(['key' => 'bar'], $options->all());

  • Declaration of Options::replace() in v2.1.0

use Symfony\Component\OptionsResolver\Options;

$options->replace([]);

Order of the given options

  • Declaration of Options::replace() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->set('firstName', 'foo');
$options->set('lastName', 'bar');
$options->replace(array('lastName' => 'bar', 'firstName' => 'foo'));

// The options are in the order of the given array, whatever their order before
static::assertSame(array('lastName' => 'bar', 'firstName' => 'foo'), $options->all());
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->set('firstName', 'foo');
$options->set('lastName', 'bar');
$options->replace(['lastName' => 'bar', 'firstName' => 'foo']);

// The options are in the order of the given array, whatever their order before
static::assertSame(['lastName' => 'bar', 'firstName' => 'foo'], $options->all());

Lazy values

A closure whose first argument is type-hinted Options is only called when the option is read, and can read the other options.

Its optional second argument, $previousValue, always receives null: replace() removes all the previous values first.

Lazy value reading an option given after it

  • Declaration of Options::replace() in v2.1.0
  • Declaration of Options::get() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->replace(array(
    'key' => function (Options $options) {
        return $options->get('firstName');
    },
    'firstName' => 'foo',
));

static::assertSame('foo', $options->get('key'));
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->replace([
    'key' => function (Options $options) {
        return $options->get('firstName');
    },
    'firstName' => 'foo',
]);

static::assertSame('foo', $options->get('key'));
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->replace([
    'key' => fn (Options $options): string => $options->get('firstName'),
    'firstName' => 'foo',
]);

static::assertSame('foo', $options->get('key'));

Normalizers kept

Unlike clear(), replace() keeps the normalizers: they apply to the new values.

Normalizer set before

  • Declaration of Options::replace() in v2.1.0
  • Declaration of Options::setNormalizer() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->setNormalizer(
    'email',
    function (Options $options, $value) {
        return strtoupper($value);
    }
);
$options->replace(array('email' => 'foo@bar.com'));

static::assertSame('FOO@BAR.COM', $options->get('email'));
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->setNormalizer(
    'email',
    function (Options $options, $value) {
        return strtoupper($value);
    }
);
$options->replace(['email' => 'foo@bar.com']);

static::assertSame('FOO@BAR.COM', $options->get('email'));
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->setNormalizer(
    'email',
    fn (Options $options, string $value): string => strtoupper($value)
);
$options->replace(['email' => 'foo@bar.com']);

static::assertSame('FOO@BAR.COM', $options->get('email'));

Option not given

The normalizer of an option not given to replace() is still called, with null: PHP reports the undefined key, and the option is added back.

  • PHP 5.3 – 7.4An E_NOTICE, Undefined index: email, then the normalizer receives null.
  • PHP >= 8.0An E_WARNING, Undefined array key "email", then the normalizer receives null.
  • Declaration of Options::replace() in v2.1.0
  • Declaration of Options::setNormalizer() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->set('email', 'foo@bar.com');
$options->setNormalizer(
    'email',
    function (Options $options, $value) {
        return $value;
    }
);
$options->replace(array('key' => 'bar'));

// E_NOTICE: Undefined index: email
static::assertSame(array('key' => 'bar', 'email' => null), $options->all());

Read-only once read

Once an option is read, the options cannot be set, overloaded, replaced, removed or cleared anymore.

Reading one option locks all of them.

Replace after count()

  • Declaration of Options::replace() in v2.1.0
  • Declaration of Options::count() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->set('key', 'foo');
count($options);
// count() does not read the options: they can still be replaced
$options->replace(array('key' => 'bar'));

static::assertSame(array('key' => 'bar'), $options->all());
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->set('key', 'foo');
count($options);
// count() does not read the options: they can still be replaced
$options->replace(['key' => 'bar']);

static::assertSame(['key' => 'bar'], $options->all());
