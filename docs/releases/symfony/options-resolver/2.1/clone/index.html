Clone an OptionsResolver

At a glance

Type
New feature
Clonable since
2.1

Since symfony/options-resolver 2.1, a cloned OptionsResolver and the original one can be changed separately.

Clone a resolver __clone()

Original resolver

PHP 5.3
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = new OptionsResolver();
$resolver->setDefaults(array('key' => 'foo'));

$clone = clone $resolver;
$clone->setDefaults(array('key' => 'bar'));

// array('key' => 'foo')
$options = $resolver->resolve();
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = (new OptionsResolver())->setDefaults(['key' => 'foo']);

$clone = clone $resolver;
$clone->setDefaults(['key' => 'bar']);

// ['key' => 'foo']
$options = $resolver->resolve();
use Symfony\Component\OptionsResolver\OptionsResolver;

$resolver = new OptionsResolver()->setDefaults(['key' => 'foo']);

$clone = clone $resolver;
$clone->setDefaults(['key' => 'bar']);

// ['key' => 'foo']
$options = $resolver->resolve();
