Options::overload()

At a glance

Type
New feature
overload() added
2.1

Since symfony/options-resolver 2.1, Options::overload() sets the value of an option, its closure receiving the previous value.

Overload an option overload()

Previous value

The previous value of the option is passed to the second argument of the closure.

  • Symfony 2.x documentation: Default Values that Depend on another Option
  • Declaration of Options::overload() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->set('roles', array('foo'));
$options->overload('roles', function (Options $options, array $previousValue) {
    return array_merge($previousValue, array('bar'));
});

// array('foo', 'bar')
$roles = $options->get('roles');
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->set('roles', ['foo']);
$options->overload('roles', function (Options $options, array $previousValue) {
    return array_merge($previousValue, ['bar']);
});

// ['foo', 'bar']
$roles = $options->get('roles');
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->set('roles', ['foo']);
$options->overload('roles', fn (Options $options, array $previousValue): array => array_merge($previousValue, ['bar']));

// ['foo', 'bar']
$roles = $options->get('roles');
