Options::overload($option, $value)

At a glance

Type
New feature
overload() added
2.1

Since symfony/options-resolver 2.1, Options::overload() sets the value of an option, its closure receiving the previous value.

Overload an option

Value

  • Declaration of Options::overload() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->overload('key', 'foo');

// array('key' => 'foo')
$all = $options->all();
use Symfony\Component\OptionsResolver\Options;

Lazy values

Lazy value

  • Symfony 2.x documentation: Default Values that Depend on another Option
  • Declaration of Options::overload() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->set('firstName', 'foo');
$options->overload(
    'key',
    // A closure whose first argument is type-hinted Options is only called when the option is read,
    // and can read the other options
    function (Options $options) {
        return $options->get('firstName');
    }
);

// 'foo'
$key = $options->get('key');
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->set('firstName', 'foo');

Previous value

  • Symfony 2.x documentation: Default Values that Depend on another Option
  • Declaration of Options::overload() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->set('roles', array('foo'));
$options->overload(
    'roles',
    // $previousValue receives the previous value of the option: array('foo')
    function (Options $options, array $previousValue) {
        return array_merge($previousValue, array('bar'));
    }
);

// array('foo', 'bar')
$roles = $options->get('roles');
use Symfony\Component\OptionsResolver\Options;

$options->set('roles', ['foo']);
$options->overload(
    'roles',

    function (Options $options, array $previousValue) {
        return array_merge($previousValue, ['bar']);
    }
);

// ['foo', 'bar']
$roles = $options->get('roles');
use Symfony\Component\OptionsResolver\Options;

Chained closures

Each closure receives the value returned by the previous one.

Previous closures discarded

A closure without a second argument discards the closures set before it: they are never called.

  • Symfony 2.x documentation: Default Values that Depend on another Option
  • Declaration of Options::overload() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->overload(
    'roles',
    function (Options $options) {
        return array('foo');
    }
);
$options->overload(
    'roles',
    function (Options $options, array $previousValue) {
        return array_merge($previousValue, array('bar'));
    }
);
$options->overload(
    'roles',
    function (Options $options) {
        return array('baz');
    }
);

// array('baz')
$roles = $options->get('roles');
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->overload(
    'roles',
    function (Options $options) {
        return ['foo'];
    }
);
$options->overload(
    'roles',
    function (Options $options, array $previousValue) {
        return array_merge($previousValue, ['bar']);
    }
);
$options->overload(
    'roles',
    function (Options $options) {
        return ['baz'];
    }
);

// ['baz']
$roles = $options->get('roles');
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->overload(
    'roles',
    fn (Options $options): array => ['foo']
);
$options->overload(
    'roles',
    fn (Options $options, array $previousValue): array => array_merge($previousValue, ['bar'])
);
$options->overload(
    'roles',
    fn (Options $options): array => ['baz']
);

// ['baz']
$roles = $options->get('roles');

Read-only once read

Once an option is read, the options cannot be set, overloaded, replaced, removed or cleared anymore.

Reading one option locks all of them.

Overload after a read

  • Declaration of Options::overload() in v2.1.0
  • Declaration of Options::get() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\Exception\OptionDefinitionException;
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->set('key', 'foo');
$options->get('key');

// OptionDefinitionException: Options cannot be overloaded anymore once options have been read.
$options->overload('key', 'bar');

  • Declaration of Options::overload() in v2.1.0
