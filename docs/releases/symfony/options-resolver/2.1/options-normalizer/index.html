Options::setNormalizer($option, \Closure $normalizer)

At a glance

Type
New feature

2.1

Since symfony/options-resolver 2.1, Options::setNormalizer() transforms the value of an option when it is read.

Normalize an option

Option normalized

  • Symfony 2.x documentation: Option Normalization
  • Declaration of Options::setNormalizer() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->set('email', 'foo@bar.com');
$options->setNormalizer(
    'email',
    function (Options $options, $value) {
        return strtoupper($value);
    }
);

// 'FOO@BAR.COM'
$email = $options->get('email');
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->set('email', 'foo@bar.com');
$options->setNormalizer(
    'email',
    fn (Options $options, string $value): string => strtoupper($value)
);

// 'FOO@BAR.COM'
$email = $options->get('email');

Value of another option

  • Symfony 2.x documentation: Option Normalization
  • Declaration of Options::setNormalizer() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->set('host', 'foo.com');
$options->set('url', '/bar');
$options->setNormalizer(
    'url',
    function (Options $options, $value) {
        return 'https://' . $options->get('host') . $value;
    }
);

// 'https://foo.com/bar'
$url = $options->get('url');
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->set('host', 'foo.com');
$options->set('url', '/bar');
$options->setNormalizer(
    'url',
    fn (Options $options, string $value): string => 'https://' . $options->get('host') . $value
);

// 'https://foo.com/bar'
$url = $options->get('url');

Normalizer added after a read

  • Symfony 2.x documentation: Option Normalization
  • Declaration of Options::setNormalizer() in v2.1.0
PHP 5.3
use Symfony\Component\OptionsResolver\Exception\OptionDefinitionException;
use Symfony\Component\OptionsResolver\Options;

$options = new Options();
$options->set('email', 'foo@bar.com');
$email = $options->get('email');

// OptionDefinitionException: Normalizers cannot be added anymore once options have been read.
$options->setNormalizer(
    'email',
    function (Options $options, $value) {
        return strtoupper($value);
    }
);
